@grist.UserTable
class Z_Libs:

  def fill_template(rec, table):
    def find_data(record):
        class FindData(dict):
            def __missing__(self,key):
                value = getattr(record, key, None)
                if value is None and '.' in key:
                    table_name, field = key.split('.',1)
                    print(table_name,field)
                    related_record = getattr(record, table_name, None)
                    if related_record:
                        value = getattr(related_record, field, None)
                return value if value is not None else f"{{{key}}}"
        return FindData()

    def fill_template(template,data):
     return template.format_map(find_data(data))

    return fill_template

  def lien_envoyer_mail(rec, table):
    from urllib.parse import quote
    def lien_envoyer_mail(titre, subject=None,to=None,cc=None,bcc=None,body=None):
      params = {}
      if subject:
        params["subject"] = subject
      if cc:
        params["cc"] = ",".join(cc)
      if bcc:
        params["bcc"] = ",".join(bcc)
      if body:
        params["body"] = body
      quotedparams = "&".join([f"{key}={quote(value)}" for key,value in params.items()])
  
      prefix = 'mailto:'
  
      return f"{titre} {prefix}{to.strip()}?{quotedparams}"
    return lien_envoyer_mail

  def C(rec, table):
    return None
